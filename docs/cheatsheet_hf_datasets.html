📥 Load a Dataset
from datasets import load_dataset

# From Hugging Face Hub

# Specific split
train = load_dataset("imdb", split="train")

# From local files
ds = load_dataset("csv",   data_files="data.csv")

ds = load_dataset("parquet", data_files="data.parquet")

🔍 Inspect & Access
ds = load_dataset("imdb")

# Structure
print(ds)                  # DatasetDict overview
print(ds["train"])        # split info
print(ds["train"].features) # column types
print(ds["train"].column_names)

# Access rows
ds["train"][0]            # first row (dict)
ds["train"][:5]           # first 5 rows
ds["train"]["label"][:5] # column slice

# Convert to pandas
df = ds["train"].to_pandas()

# Convert from pandas
from datasets import Dataset
ds = Dataset.from_pandas(df)
🗺️ map() — Transform Data
# map() is the core transformation tool
from transformers import AutoTokenizer
tokenizer = AutoTokenizer.from_pretrained("bert-base-uncased")

def tokenize(examples):
    return tokenizer(examples["text"],
                      truncation=True,
                      padding="max_length",
                      max_length=128)

# batched=True = fast (vectorized)
ds = ds.map(tokenize, batched=True)

# Add a new column
ds = ds.map(lambda x: {"len": len(x["text"])})

# Remove columns during map
ds = ds.map(tokenize, batched=True,
             remove_columns=["text"])

# Cache control
ds = ds.map(tokenize, batched=True,
             load_from_cache_file=False)
🔎 Filter, Select & Sort
# Filter rows by condition
ds = ds.filter(lambda x: x["label"] == 1)
ds = ds.filter(lambda x: len(x["text"]) > 100)

# Select specific indices
ds = ds.select(range(1000))    # first 1000
ds = ds.select([0, 5, 10])     # specific rows

# Shuffle
ds = ds.shuffle(seed=42)

# Sort
ds = ds.sort("label")

# Column ops
ds = ds.rename_column("label", "labels")
ds = ds.remove_columns(["text"])
ds = ds.select_columns(["input_ids", "labels"])
✂️ Split & Combine
# Train/test split from a single dataset
split = ds.train_test_split(test_size=0.2, seed=42)
train_ds = split["train"]
test_ds  = split["test"]

# Concatenate datasets
from datasets import concatenate_datasets
full = concatenate_datasets([ds1, ds2])

# Interleave (mix multiple datasets)
from datasets import interleave_datasets
mixed = interleave_datasets([ds1, ds2],
                               probabilities=[0.7, 0.3])

# Create from dict
from datasets import Dataset
ds = Dataset.from_dict({
    "text": ["hello", "world"],
    "label": [0, 1],
})
🔄 Format for PyTorch
# Set format → tensors on the fly
ds.set_format(type="torch",
              columns=["input_ids", "attention_mask", "labels"])

# or: with_format (returns new dataset)
ds = ds.with_format("torch")

# Then wrap in DataLoader
from torch.utils.data import DataLoader
loader = DataLoader(ds, batch_size=32, shuffle=True)

# Apply transforms lazily (no copy)
ds = ds.with_transform(transform_fn)

# Reset to Python dicts
ds.reset_format()
⚠️ Always set_format AFTER all map/filter ops, not before.

💡 Essential Tips
# Check dataset size before loading everything
ds = load_dataset("bigdata", streaming=True)

# Speed up map() with multiprocessing
ds = ds.map(fn, batched=True, num_proc=4)

# Cast column type
from datasets import ClassLabel, Value
ds = ds.cast_column("label", ClassLabel(names=["neg","pos"]))

# Unique values
ds.unique("label")

# Dataset info
ds.info      # metadata
ds.num_rows  # row count
len(ds)      # same
💡 batched=True in map() is much faster (10-100x with fast tokenizers; ~9x even on a tiny one) — always use it.
