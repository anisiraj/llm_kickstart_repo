✏️ Tokenizer
from transformers import AutoTokenizer

tok = AutoTokenizer.from_pretrained("bert-base-uncased")

# Encode single sentence
enc = tok("Hello world",
          return_tensors="pt",
          truncation=True,
          padding="max_length",
          max_length=128)
# → {input_ids, attention_mask, token_type_ids}

# Batch encode
enc = tok(["Hello", "World"],
          padding=True, truncation=True,
          return_tensors="pt")

# Decode back to text
tok.decode(enc["input_ids"][0])
tok.batch_decode(enc["input_ids"],
                  skip_special_tokens=True)

# Chat template (instruct models)
msgs = [{"role":"user", "content":"Hi!"}]
tok.apply_chat_template(msgs, tokenize=False)

    load_in_4bit=True,      # QLoRA memory saving
)

# AutoModelForAudioClassification  → audio

🏋️ Trainer — Fine-Tune a Model
from transformers import (
    AutoModelForSequenceClassification, AutoTokenizer,
    TrainingArguments, Trainer, DataCollatorWithPadding
)
from datasets import load_dataset

model = AutoModelForSequenceClassification.from_pretrained(
    "distilbert-base-uncased", num_labels=2)
tok   = AutoTokenizer.from_pretrained("distilbert-base-uncased")
ds    = load_dataset("imdb")
ds    = ds.map(lambda x: tok(x["text"], truncation=True), batched=True)

args = TrainingArguments(
    output_dir="./output",
    num_train_epochs=3,
    per_device_train_batch_size=16,
    per_device_eval_batch_size=16,
    learning_rate=2e-5,
    eval_strategy="epoch",
    save_strategy="epoch",
    load_best_model_at_end=True,
    fp16=True,
    push_to_hub=False,
)

trainer = Trainer(model=model, args=args,
    train_dataset=ds["train"],
    eval_dataset=ds["test"],
    processing_class=tok,
    data_collator=DataCollatorWithPadding(tok))
trainer.train()

📊 Evaluation & Metrics
import evaluate
import numpy as np

metric = evaluate.load("accuracy")
# Other: "f1", "rouge", "bleu", "seqeval"

def compute_metrics(eval_pred):
    logits, labels = eval_pred
    preds = np.argmax(logits, axis=-1)
    return metric.compute(
        predictions=preds, references=labels)

# Pass to Trainer
trainer = Trainer(
    ...,
    compute_metrics=compute_metrics,
)

# Run evaluation only
trainer.evaluate()

💡 Key Tips & Common Patterns
# Always use Auto classes → portable across models
# AutoTokenizer, AutoModel, AutoConfig

# Key TrainingArguments to know
# fp16=True          → half precision (saves VRAM)
# gradient_accumulation_steps=4  → fake larger batch
# warmup_ratio=0.1   → LR warmup for 10% of steps
# weight_decay=0.01  → L2 regularization
# logging_steps=50   → print loss every 50 steps

# Freeze base, train head only
for param in model.base_model.parameters():
    param.requires_grad = False

# Config: see hyperparams
print(model.config)

# Move model to GPU
model = model.to("cuda")
# or device_map="auto" at from_pretrained
⚠️ Use eval_strategy="epoch" not evaluation_strategy — renamed in v4.38+ 💡 Trainer handles device placement automatically — no need to .to(device) manually.
