📝 Training Script (train.py)
# scripts/train.py — runs INSIDE the container
import argparse, os
from transformers import (AutoModelForSequenceClassification,
    AutoTokenizer, Trainer, TrainingArguments)
from datasets import load_from_disk

parser = argparse.ArgumentParser()
parser.add_argument("--model_name", default="distilbert-base-uncased")
parser.add_argument("--epochs", type=int, default=3)
parser.add_argument("--lr", type=float, default=2e-5)
parser.add_argument("--train_dir",
    default=os.environ["SM_CHANNEL_TRAIN"])  # ← SageMaker injects
parser.add_argument("--model_dir",
    default=os.environ["SM_MODEL_DIR"])      # ← save here
args = parser.parse_args()

model     = AutoModelForSequenceClassification.from_pretrained(args.model_name, num_labels=2)
tokenizer = AutoTokenizer.from_pretrained(args.model_name)
dataset   = load_from_disk(args.train_dir)

trainer = Trainer(model=model,
    args=TrainingArguments(args.model_dir, num_train_epochs=args.epochs,
        learning_rate=args.lr, per_device_train_batch_size=16),
    train_dataset=dataset)
trainer.train()
model.save_pretrained(args.model_dir)  # ← MUST save to SM_MODEL_DIR
tokenizer.save_pretrained(args.model_dir)

    transformers_version= "4.36",
    pytorch_version     = "2.1",
    py_version          = "py310",
    hyperparameters     = {
        "model_name": "distilbert-base-uncased",
        "epochs": 3, "lr": 2e-5,
    },
    # Spot instance = up to 70% cheaper
    use_spot_instances  = True,
    max_wait            = 7200,
    max_run             = 3600,
    checkpoint_s3_uri   = f"s3://{bucket}/checkpoints",
)

estimator.fit({"train": train_s3})  # blocks until done
print(estimator.model_data)           # S3 path to model.tar.gz

# bedrock = managed FM inference, no infra
import boto3, json

bedrock_rt = boto3.client("bedrock-runtime",
                            region_name="us-east-1")

# Invoke any foundation model (Converse API — universal)
response = bedrock_rt.converse(
    modelId  = "anthropic.claude-3-5-sonnet-20241022-v2:0",
    messages = [{
        "role": "user",
        "content": [{"text": "Summarize transformers in 3 lines."}]
    }],
    inferenceConfig = {"maxTokens": 512, "temperature": 0.7}
)
print(response["output"]["message"]["content"][0]["text"])

# Streaming
stream = bedrock_rt.converse_stream(modelId="...", messages=[...])
for event in stream["stream"]:
    if "contentBlockDelta" in event:
        print(event["contentBlockDelta"]["delta"]["text"], end="")

🎯 Bedrock — Custom Model Fine-Tuning
# 1. Data format: JSONL in S3
# {"prompt": "...", "completion": "..."}

bedrock = boto3.client("bedrock", region_name="us-east-1")

# 2. Create fine-tuning job
resp = bedrock.create_model_customization_job(
    customizationJobName = "my-ft-job",
    customModelName      = "my-custom-llama",
    baseModelIdentifier  = "meta.llama3-8b-instruct-v1:0",
    roleArn              = "arn:aws:iam::ACCT:role/BedrockRole",
    trainingDataConfig   = {"s3Uri": "s3://bucket/train.jsonl"},
    outputDataConfig     = {"s3Uri": "s3://bucket/output/"},
    hyperParameters      = {
        "epochCount": "1", "batchSize": "8",
        "learningRate": "0.00001"},
)

# 3. Check status
job = bedrock.get_model_customization_job(
    jobIdentifier=resp["jobArn"])
print(job["status"])                   # InProgress / Completed

# 4. Provision throughput to invoke custom model
pt = bedrock.create_provisioned_model_throughput(
    modelUnits          = 1,
    provisionedModelName= "my-prod-model",
    modelId             = job["outputModelArn"])
# → use pt["provisionedModelArn"] in converse(modelId=...)

✅ SageMaker = full control, bring your own code, any HF model.
✅ Bedrock = no infra, pay-per-token, managed FMs only.
⚡ SageMaker vs Bedrock — When to Use What
🔵 Use SageMaker when:
• Training / fine-tuning your own model
• Deploying any HF Hub model with custom code
• Need full control over infra, GPU, distributed training
• Running batch transform jobs on large datasets
🟢 Use Bedrock when:
• You just need to call a Foundation Model via API
• Fine-tuning on supported base models (no infra setup)
• Building apps fast — no container management
• Pay-per-token pricing with no idle endpoint costs
